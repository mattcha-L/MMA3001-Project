optimised_neural_network

Incrementally train one neural network week by week and evaluate it on a test week.

A single MLPRegressor is updated with partial_fit(), one week at a time in chronological order, so the most recent week is always the last update applied.

Week roles: TRIAL_WEEKS: Trained on in order; each week calls partial_fit(). VALIDATION_WEEKS: Never trained on; scored during training for monitoring. TEST_WEEKS: Never trained on; used only for the final evaluation.

Features: elapsed time plus sin/cos of hour-of-day and day-of-week.

Efficiency notes: * Readings live only in the master DataFrame data. * One compact (t, y) array pair and one scaled feature matrix X are built; every week or split is a slice (view) of them. * The scaler is fitted on training slices only, so there is no leakage.

Settings: VARIABLE_TO_MODEL, TRIAL_WEEKS, VALIDATION_WEEKS, TEST_WEEKS, EPOCHS_PER_WEEK, VALIDATE_EVERY

 85def _prepare(variable_key, weekly):
 86    """Builds the compact dataset for one variable.
 87
 88    Drops NaN values and duplicate timestamps, then maps each week's master
 89    slice onto the compacted arrays.
 90
 91    Args:
 92        variable_key: Column name in `data`.
 93        weekly: Dict of week start -> slice into `data`.
 94
 95    Returns:
 96        A Dataset with t, y and per-week row slices.
 97    """
 98    t_all = _epoch_seconds()
 99    y_all = data[variable_key].to_numpy(dtype=float)
100
101    valid = np.flatnonzero(~np.isnan(y_all))
102    tv = t_all[valid]
103    # Data is time-sorted, so duplicate timestamps are adjacent
104    dup = np.zeros(len(valid), dtype=bool)
105    dup[1:] = tv[1:] == tv[:-1]
106    keep_pos = valid[~dup]                      # master row positions that survive
107
108    # Convert each week's master slice into a slice of the compacted arrays
109    rows = {}
110    for week, sl in weekly.items():
111        a = int(np.searchsorted(keep_pos, sl.start, side="left"))
112        b = int(np.searchsorted(keep_pos, sl.stop, side="left"))
113        rows[week] = slice(a, b)
114    return Dataset(t=t_all[keep_pos], y=y_all[keep_pos], rows=rows)

117def _build_features(t, t0):
118    """Builds the 5-column feature matrix from Unix seconds.
119
120    Args:
121        t: Timestamps as Unix seconds.
122        t0: Reference time that elapsed time is measured from.
123
124    Returns:
125        Array of shape (len(t), 5): elapsed time, sin/cos(hour), sin/cos(day of week).
126    """
127    X = np.empty((len(t), 5))
128    hour = ((t % 86400) // 60) / 60              # hour + minute/60
129    dow = ((t // 86400 + 3) % 7) + hour / 24     # 1970-01-01 was a Thursday (Mon=0 -> 3)
130    X[:, 0] = t - t0
131    X[:, 1] = np.sin(2 * np.pi * hour / 24)
132    X[:, 2] = np.cos(2 * np.pi * hour / 24)
133    X[:, 3] = np.sin(2 * np.pi * dow / 7)
134    X[:, 4] = np.cos(2 * np.pi * dow / 7)
135    return X

138def _resolve_weeks(weeks_spec, weekly, week_numbers):
139    """Converts a week specification into week-start Timestamps.
140
141    Args:
142        weeks_spec: None, "all", "last", a week number or date, or a list of those.
143        weekly: Dict of week start -> slice.
144        week_numbers: Dict of week number -> week start.
145
146    Returns:
147        Sorted list of week-start Timestamps (unknown weeks are warned about and skipped).
148    """
149    if weeks_spec is None:
150        return []
151    if isinstance(weeks_spec, str) and weeks_spec == "all":
152        return sorted(weekly)
153    if isinstance(weeks_spec, str) and weeks_spec == "last":
154        return [week_numbers[max(week_numbers)]] if week_numbers else []
155    items = [weeks_spec] if isinstance(weeks_spec, (int, str, pd.Timestamp)) else list(weeks_spec)
156    resolved = set()
157    for item in items:
158        if isinstance(item, int):
159            week = week_numbers.get(item)
160            if week is None:
161                print(f"[warning] week number {item} not found")
162            else:
163                resolved.add(week)
164        else:
165            ts = pd.Timestamp(item)
166            if ts in weekly:
167                resolved.add(ts)
168            else:
169                print(f"[warning] week {item} not found")
170    return sorted(resolved)

173def _runs(weeks, rows):
174    """Converts weeks into slices, merging weeks that touch end-to-end.
175
176    Args:
177        weeks: Week starts in chronological order.
178        rows: Dict of week start -> slice.
179
180    Returns:
181        List of slices; empty weeks are skipped.
182    """
183    runs = []
184    for w in weeks:
185        sl = rows[w]
186        if sl.stop == sl.start:
187            continue
188        if runs and runs[-1].stop == sl.start:
189            runs[-1] = slice(runs[-1].start, sl.stop)    # extend the previous run
190        else:
191            runs.append(sl)
192    return runs

195def _take(arr, runs):
196    """Selects the rows covered by `runs`.
197
198    Args:
199        arr: Array to select from.
200        runs: List of slices.
201
202    Returns:
203        A view if there is one run; otherwise a small concatenated copy.
204    """
205    if not runs:
206        return arr[:0]
207    if len(runs) == 1:
208        return arr[runs[0]]
209    return np.concatenate([arr[r] for r in runs])

212def _thinned(arr, runs, max_points):
213    """Takes a strided sample of `arr` over `runs` (for plotting).
214
215    The same stride is used for every array built from the same runs, so
216    thinned x and y values stay aligned.
217
218    Args:
219        arr: Array to sample.
220        runs: List of slices.
221        max_points: Approximate maximum number of points; None for no thinning.
222
223    Returns:
224        The thinned array.
225    """
226    total = sum(r.stop - r.start for r in runs)
227    stride = max(1, math.ceil(total / max_points)) if max_points else 1
228    parts = [arr[r.start:r.stop:stride] for r in runs]
229    return np.concatenate(parts) if parts else arr[:0]

232def _as_datetime(t):
233    """Converts Unix seconds to datetime64[s] (UTC), for plotting only."""
234    return np.rint(t).astype("int64").astype("datetime64[s]")

237def _error_metrics(y_true, y_pred):
238    """Computes error metrics.
239
240    Args:
241        y_true: Observed values.
242        y_pred: Predicted values.
243
244    Returns:
245        Dict with "rmse", "rrmse" (RMSE divided by the RMS of y_true) and "mae".
246    """
247    residuals = y_true - y_pred
248    rms = np.sqrt(np.mean(y_true**2))
249    rmse = np.sqrt(np.mean(residuals ** 2))
250    rrmse = np.sqrt(np.mean(residuals ** 2))/rms
251    mae = np.mean(np.abs(residuals))
252    return {"rmse": rmse, "rrmse": rrmse, "mae": mae }

258def train_incremental(variable_key=VARIABLE_TO_MODEL, trial_weeks=TRIAL_WEEKS,
259                      validation_weeks=VALIDATION_WEEKS, test_weeks=TEST_WEEKS,
260                      min_points_per_week=MIN_POINTS_PER_WEEK, epochs_per_week=EPOCHS_PER_WEEK,
261                      validate_every=VALIDATE_EVERY, weekly=None, week_numbers=None):
262    """Trains one MLPRegressor week by week and evaluates it on the test weeks.
263
264    Args:
265        variable_key: Column name in `data` (see VARIABLES).
266        trial_weeks: Weeks to train on, or "all_except_holdout".
267        validation_weeks: Weeks used only to monitor progress.
268        test_weeks: Weeks used only for the final evaluation.
269        min_points_per_week: Weeks with fewer readings are skipped.
270        epochs_per_week: partial_fit passes per week.
271        validate_every: Score validation every N trained weeks.
272        weekly: Precomputed week slices; computed here if None.
273        week_numbers: Precomputed week numbering; computed here if None.
274
275    Returns:
276        Dict with the model, scaler, validation history, dataset, slices and (if
277        test data exists) test predictions and metrics; None if nothing could be trained.
278    """
279    display_name, unit = VARIABLES[variable_key]
280
281    if weekly is None:
282        weekly = group_by_week(data)
283    if week_numbers is None:
284        week_numbers = number_weeks(weekly)
285    number_of = {w: n for n, w in week_numbers.items()}        # week start -> week number
286
287    # ---- split weeks into train / validation / test ----
288    val_weeks = _resolve_weeks(validation_weeks, weekly, week_numbers)
289    test_weeks_ = _resolve_weeks(test_weeks, weekly, week_numbers)
290    holdout = set(val_weeks) | set(test_weeks_)
291
292    if isinstance(trial_weeks, str) and trial_weeks == "all_except_holdout":
293        train_weeks = sorted(w for w in week_numbers.values() if w not in holdout)
294    else:
295        candidate = _resolve_weeks(trial_weeks, weekly, week_numbers)
296        overlap = [w for w in candidate if w in holdout]
297        if overlap:
298            print(f"[warning] {display_name}: {len(overlap)} week(s) requested for training also "
299                  f"appear in validation/test -- removing from training to avoid leakage.")
300        train_weeks = [w for w in candidate if w not in holdout]
301
302    # ---- compact arrays for this variable (once) ----
303    ds = _prepare(variable_key, weekly)
304    rows = ds.rows
305
306    train_runs = _runs(train_weeks, rows)
307    val_runs = _runs(val_weeks, rows)
308    test_runs = _runs(test_weeks_, rows)
309    if not train_runs:
310        print(f"[error] {display_name}: no training data available -- check TRIAL_WEEKS/TEST_WEEKS.")
311        return None
312
313    # ---- ONE feature matrix, ONE scaler fit (on training slices only), scaled in place ----
314    t0 = ds.t[train_runs[0].start]                       # data is time-sorted => earliest training point
315    X = _build_features(ds.t, t0)
316    scaler = StandardScaler()
317    for run in train_runs:
318        scaler.partial_fit(X[run])
319    scaler.transform(X, copy=False)
320
321    # Validation / test sets sliced once, outside the training loop.
322    X_val, y_val = _take(X, val_runs), _take(ds.y, val_runs)
323    X_test, y_test = _take(X, test_runs), _take(ds.y, test_runs)
324
325    model = MLPRegressor(activation=NN_ACTIVATION, hidden_layer_sizes=NN_HIDDEN_LAYERS, learning_rate_init=NN_ALPHA, random_state=RANDOM_SEED)
326
327    # Keep only weeks with enough data.
328    trainable = []
329    for week_start in train_weeks:
330        n = rows[week_start].stop - rows[week_start].start
331        if n < min_points_per_week:
332            print(f"[skip] {display_name}: week {number_of[week_start]} ({week_start.date()}) has only "
333                  f"{n} point(s), need at least {min_points_per_week} -- skipping.")
334        else:
335            trainable.append(week_start)
336    if not trainable:
337        print(f"[error] {display_name}: no week had enough data to train on.")
338        return None
339
340    # ---- train week by week, updating the same weights ----
341    validation_history = []
342    for i, week_start in enumerate(trainable, start=1):
343        week_number = number_of[week_start]
344        sl = rows[week_start]
345        X_week, y_week = X[sl], ds.y[sl]                 # views, no copies
346
347        t_fit0 = time.perf_counter()
348        for _ in range(epochs_per_week):
349            model.partial_fit(X_week, y_week)
350        fit_seconds = time.perf_counter() - t_fit0
351
352        msg = (f"{display_name}: week {week_number} ({week_start.date()}) -- updated weights with "
353               f"{len(y_week)} point(s) over {epochs_per_week} epochs in {fit_seconds:.2f}s")
354        # Score validation every `validate_every` weeks, and always after the last week
355        if len(y_val) > 0 and (i % validate_every == 0 or i == len(trainable)):
356            val_metrics = _error_metrics(y_val, model.predict(X_val))
357            validation_history.append({"week_number": week_number, "week_start": week_start, **val_metrics})
358            msg += f" | running validation RMSE {val_metrics['rmse']:.3f} {unit}, rRMSE {val_metrics['rrmse']:.3f}"
359        print(msg)
360
361    results = {
362        "variable_key": variable_key, "display_name": display_name, "unit": unit,
363        "model": model, "scaler": scaler, "t0": t0, "n_weeks_trained": len(trainable),
364        "validation_history": validation_history,
365        # slices + the single compact dataset, instead of DataFrame copies
366        "dataset": ds, "train_runs": train_runs, "val_runs": val_runs, "test_runs": test_runs,
367    }
368
369    # ---- final evaluation on the held-out test weeks ----
370    if len(y_test) > 0:
371        y_pred = model.predict(X_test)
372        metrics = _error_metrics(y_test, y_pred)
373        results.update(test_t=_as_datetime(_take(ds.t, test_runs)), test_y=y_test,
374                       test_pred=y_pred, **metrics)
375        rrmse_str = "n/a" if np.isnan(metrics["rrmse"]) else f"{metrics['rrmse']:.3f}"
376        print(f"{display_name}: FINAL model evaluated on {len(y_test)} test point(s) -- "
377              f"RMSE {metrics['rmse']:.3f} {unit}, MAE {metrics['mae']:.3f} {unit}, rRMSE {rrmse_str}")
378    else:
379        print(f"[warning] {display_name}: no test data available -- check TEST_WEEKS")
380
381    return results

387def plot_incremental_evaluation(results):
388    """Plots test performance: time series (left) and predicted vs. true (right).
389
390    The left panel shows the test week against the model prediction, with faint
391    training and validation data as context.
392
393    Args:
394        results: Dict returned by train_incremental().
395
396    Returns:
397        The matplotlib Figure, or None if there is no test prediction.
398    """
399    if results is None or "test_pred" not in results:
400        return None
401    ds = results["dataset"]
402    display_name, unit = results["display_name"], results["unit"]
403    test_t, y_true, y_pred = results["test_t"], results["test_y"], results["test_pred"]
404
405    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(16, 5.5), gridspec_kw={"width_ratios": [2, 1]})
406
407    # Faint context: training and validation data
408    ax1.scatter(_as_datetime(_thinned(ds.t, results["train_runs"], PLOT_MAX_POINTS)),
409                _thinned(ds.y, results["train_runs"], PLOT_MAX_POINTS),
410                s=10, alpha=0.3, color="steelblue", label="Training data (all weeks)")
411    if results["val_runs"]:
412        ax1.scatter(_as_datetime(_thinned(ds.t, results["val_runs"], PLOT_MAX_POINTS)),
413                    _thinned(ds.y, results["val_runs"], PLOT_MAX_POINTS),
414                    s=10, alpha=0.3, color="green", label="Validation data")
415
416    # Test data and model prediction
417    step = max(1, math.ceil(len(y_true) / PLOT_MAX_POINTS))   # test arrays are already time-sorted
418    ax1.scatter(test_t[::step], y_true[::step], s=20, alpha=0.6, color="crimson", marker="s",
419                label="True test data")
420    ax1.plot(test_t, y_pred, color="darkorange", linewidth=2.5, label="Model prediction")
421
422    ax1.set_title(f"{display_name} -- incrementally trained over {results['n_weeks_trained']} week(s)")
423    ax1.set_xlabel("Timestamp")
424    ax1.set_ylabel(f"{display_name} ({unit})")
425    ax1.grid(True, alpha=0.3, linestyle="--")
426    ax1.legend(fontsize=9)
427    ax1.tick_params(axis="x", rotation=30)
428
429    # Right panel: points on the dashed line are perfect predictions
430    lims = [min(y_true.min(), y_pred.min()), max(y_true.max(), y_pred.max())]
431    ax2.plot(lims, lims, "k--", linewidth=1, label="Perfect prediction")
432    ax2.scatter(y_true, y_pred, s=20, alpha=0.5, color="crimson")
433    ax2.set_xlabel(f"True {display_name} ({unit})")
434    ax2.set_ylabel(f"Predicted {display_name} ({unit})")
435    ax2.set_title(f"Test set: rRMSE={results['rrmse']*100:.3f}%, RMSE={results['rmse']:.2f}")
436    ax2.grid(True, alpha=0.3, linestyle="--")
437    ax2.legend(fontsize=9)
438
439    fig.tight_layout()
440    return fig

259def train_incremental(variable_key=VARIABLE_TO_MODEL, trial_weeks=TRIAL_WEEKS,
260                      validation_weeks=VALIDATION_WEEKS, test_weeks=TEST_WEEKS,
261                      min_points_per_week=MIN_POINTS_PER_WEEK, epochs_per_week=EPOCHS_PER_WEEK,
262                      validate_every=VALIDATE_EVERY, weekly=None, week_numbers=None):
263    """Trains one MLPRegressor week by week and evaluates it on the test weeks.
264
265    Args:
266        variable_key: Column name in `data` (see VARIABLES).
267        trial_weeks: Weeks to train on, or "all_except_holdout".
268        validation_weeks: Weeks used only to monitor progress.
269        test_weeks: Weeks used only for the final evaluation.
270        min_points_per_week: Weeks with fewer readings are skipped.
271        epochs_per_week: partial_fit passes per week.
272        validate_every: Score validation every N trained weeks.
273        weekly: Precomputed week slices; computed here if None.
274        week_numbers: Precomputed week numbering; computed here if None.
275
276    Returns:
277        Dict with the model, scaler, validation history, dataset, slices and (if
278        test data exists) test predictions and metrics; None if nothing could be trained.
279    """
280    display_name, unit = VARIABLES[variable_key]
281
282    if weekly is None:
283        weekly = group_by_week(data)
284    if week_numbers is None:
285        week_numbers = number_weeks(weekly)
286    number_of = {w: n for n, w in week_numbers.items()}        # week start -> week number
287
288    # ---- split weeks into train / validation / test ----
289    val_weeks = _resolve_weeks(validation_weeks, weekly, week_numbers)
290    test_weeks_ = _resolve_weeks(test_weeks, weekly, week_numbers)
291    holdout = set(val_weeks) | set(test_weeks_)
292
293    if isinstance(trial_weeks, str) and trial_weeks == "all_except_holdout":
294        train_weeks = sorted(w for w in week_numbers.values() if w not in holdout)
295    else:
296        candidate = _resolve_weeks(trial_weeks, weekly, week_numbers)
297        overlap = [w for w in candidate if w in holdout]
298        if overlap:
299            print(f"[warning] {display_name}: {len(overlap)} week(s) requested for training also "
300                  f"appear in validation/test -- removing from training to avoid leakage.")
301        train_weeks = [w for w in candidate if w not in holdout]
302
303    # ---- compact arrays for this variable (once) ----
304    ds = _prepare(variable_key, weekly)
305    rows = ds.rows
306
307    train_runs = _runs(train_weeks, rows)
308    val_runs = _runs(val_weeks, rows)
309    test_runs = _runs(test_weeks_, rows)
310    if not train_runs:
311        print(f"[error] {display_name}: no training data available -- check TRIAL_WEEKS/TEST_WEEKS.")
312        return None
313
314    # ---- ONE feature matrix, ONE scaler fit (on training slices only), scaled in place ----
315    t0 = ds.t[train_runs[0].start]                       # data is time-sorted => earliest training point
316    X = _build_features(ds.t, t0)
317    scaler = StandardScaler()
318    for run in train_runs:
319        scaler.partial_fit(X[run])
320    scaler.transform(X, copy=False)
321
322    # Validation / test sets sliced once, outside the training loop.
323    X_val, y_val = _take(X, val_runs), _take(ds.y, val_runs)
324    X_test, y_test = _take(X, test_runs), _take(ds.y, test_runs)
325
326    model = MLPRegressor(activation=NN_ACTIVATION, hidden_layer_sizes=NN_HIDDEN_LAYERS, learning_rate_init=NN_ALPHA, random_state=RANDOM_SEED)
327
328    # Keep only weeks with enough data.
329    trainable = []
330    for week_start in train_weeks:
331        n = rows[week_start].stop - rows[week_start].start
332        if n < min_points_per_week:
333            print(f"[skip] {display_name}: week {number_of[week_start]} ({week_start.date()}) has only "
334                  f"{n} point(s), need at least {min_points_per_week} -- skipping.")
335        else:
336            trainable.append(week_start)
337    if not trainable:
338        print(f"[error] {display_name}: no week had enough data to train on.")
339        return None
340
341    # ---- train week by week, updating the same weights ----
342    validation_history = []
343    for i, week_start in enumerate(trainable, start=1):
344        week_number = number_of[week_start]
345        sl = rows[week_start]
346        X_week, y_week = X[sl], ds.y[sl]                 # views, no copies
347
348        t_fit0 = time.perf_counter()
349        for _ in range(epochs_per_week):
350            model.partial_fit(X_week, y_week)
351        fit_seconds = time.perf_counter() - t_fit0
352
353        msg = (f"{display_name}: week {week_number} ({week_start.date()}) -- updated weights with "
354               f"{len(y_week)} point(s) over {epochs_per_week} epochs in {fit_seconds:.2f}s")
355        # Score validation every `validate_every` weeks, and always after the last week
356        if len(y_val) > 0 and (i % validate_every == 0 or i == len(trainable)):
357            val_metrics = _error_metrics(y_val, model.predict(X_val))
358            validation_history.append({"week_number": week_number, "week_start": week_start, **val_metrics})
359            msg += f" | running validation RMSE {val_metrics['rmse']:.3f} {unit}, rRMSE {val_metrics['rrmse']:.3f}"
360        print(msg)
361
362    results = {
363        "variable_key": variable_key, "display_name": display_name, "unit": unit,
364        "model": model, "scaler": scaler, "t0": t0, "n_weeks_trained": len(trainable),
365        "validation_history": validation_history,
366        # slices + the single compact dataset, instead of DataFrame copies
367        "dataset": ds, "train_runs": train_runs, "val_runs": val_runs, "test_runs": test_runs,
368    }
369
370    # ---- final evaluation on the held-out test weeks ----
371    if len(y_test) > 0:
372        y_pred = model.predict(X_test)
373        metrics = _error_metrics(y_test, y_pred)
374        results.update(test_t=_as_datetime(_take(ds.t, test_runs)), test_y=y_test,
375                       test_pred=y_pred, **metrics)
376        rrmse_str = "n/a" if np.isnan(metrics["rrmse"]) else f"{metrics['rrmse']:.3f}"
377        print(f"{display_name}: FINAL model evaluated on {len(y_test)} test point(s) -- "
378              f"RMSE {metrics['rmse']:.3f} {unit}, MAE {metrics['mae']:.3f} {unit}, rRMSE {rrmse_str}")
379    else:
380        print(f"[warning] {display_name}: no test data available -- check TEST_WEEKS")
381
382    return results
